// 11-sep-2026 · Entra el AGENTE DE CLAUDE CODE, con lectura y una bóveda.
//   Es un usuario común de Authentication —no una cuenta de servicio, que
//   saltearía estas reglas enteras— y existe para que un chat pueda comparar
//   lo que hay en la base con lo que dice el código publicado: sobre todo
//   `sitio/publicado` contra `contenido.json` y los textos de `index.html`.
//
//   Lee todo MENOS `calculos` e `invitaciones`. Los cálculos llevan datos de
//   clientes y medidas de obra, y detrás costos, márgenes y tarifas; las
//   invitaciones son mails de gente que todavía no entró. De escritura tiene
//   exactamente un documento, `sitio/taller`, desde el 21-sep-2026 — ver la
//   nota de arriba.
//
//   NO tiene ficha en `usuarios/`, a propósito: no es del equipo, no tiene rol
//   ni permisos, y no aparece en la pantalla de usuarios. Su acceso sale de
//   ese bloque y nada más.
//
//   La regla compara el UID y no el mail: Firebase deja que cualquiera se
//   registre con mail y contraseña, así que una regla que abre la base «a
//   quien tenga tal mail» se la abre a quien consiga registrar ese mail
//   primero. El UID lo asigna Firebase y no se puede elegir. Es distinto en
//   cada proyecto aunque la cuenta sea la misma.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- quién es quién ----------
    // Se lee el documento usuarios/{uid} para saber el rol. Los permisos son
    // datos, no reglas: se cambian desde el panel sin republicar esto.

    function yo() {
      return get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data;
    }
    function logueado() {
      return request.auth != null
             && exists(/databases/$(database)/documents/usuarios/$(request.auth.uid));
    }
    function activo() {
      return logueado() && yo().activo == true;
    }
    function admin() {
      return activo() && yo().rol == 'admin';
    }
    // Un permiso concreto. El administrador los tiene todos por definición.
    function con(p) {
      return activo() && (yo().rol == 'admin' || yo().permisos[p] == true);
    }
    function inv() {
      return get(/databases/$(database)/documents/invitaciones/$(request.auth.token.email.lower())).data;
    }

    // ---------- el agente de Claude Code ----------
    // Ver la nota de arriba. Lectura sola, con dos colecciones afuera. La
    // lista de abajo tiene que decir lo mismo que `selladas` del proyecto
    // `casayourte` en datos/herramientas/firestore.mjs: el archivo da el
    // mensaje claro, ESTA REGLA da la garantía. Si cambia una, cambia la otra
    // en la misma tanda.
    function esAgente() {
      return request.auth != null
             && request.auth.uid == 'MwuvZZQGd0WPWU8ia65dZBycEC43';
    }

    match /{coleccion}/{resto=**} {
      allow read: if esAgente() && !(coleccion in ['calculos', 'invitaciones', 'avisos_contacto']);
    }

    // ---------- usuarios ----------
    // Modelo: UNA cuenta admin, y colaboradores con permisos explícitos.
    // El rol y los permisos son DATOS: cambiarlos no requiere republicar esto.

    match /usuarios/{uid} {

      // Cada uno lee el suyo. El administrador, todos.
      allow read: if request.auth != null && (request.auth.uid == uid || admin());

      // Alta por invitación: sólo TU propio documento, con TU mail, si existe
      // invitación para ese mail, y con EL ROL Y LOS PERMISOS que dice la
      // invitación. Esto impide que alguien se dé permisos a sí mismo.
      allow create: if request.auth != null
        && request.auth.uid == uid
        && request.resource.data.email == request.auth.token.email.lower()
        && request.resource.data.activo == true
        && exists(/databases/$(database)/documents/invitaciones/$(request.auth.token.email.lower()))
        && request.resource.data.rol == inv().rol
        && request.resource.data.permisos == inv().permisos;

      // Sólo el administrador cambia roles, permisos y activo. Y no puede
      // quitarse a sí mismo el rol ni desactivarse: es la forma más rápida de
      // quedarse afuera del propio panel sin manera de volver.
      allow update: if admin()
        && (uid != request.auth.uid
            || (request.resource.data.rol == resource.data.rol
                && request.resource.data.activo == resource.data.activo));

      allow delete: if admin() && uid != request.auth.uid;
    }

    // ---------- invitaciones ----------
    // El id del documento es el mail invitado, en minúsculas. Lleva el rol y
    // los permisos que va a recibir quien se registre con ese mail.
    match /invitaciones/{mail} {
      // La persona invitada necesita leer la suya ANTES de tener permisos.
      allow read: if admin()
                  || (request.auth != null && request.auth.token.email.lower() == mail);
      allow create, update: if admin();
      // Se borra sola al usarse, o la anula el administrador.
      allow delete: if admin()
                    || (request.auth != null && request.auth.token.email.lower() == mail);
    }

    // ---------- álbumes y fotos ----------
    // Desde el 19-oct-2026 el AGENTE también escribe los álbumes y sus fotos
    // (Mauro: «sí, también los álbumes, pero quiero que puedas crear y
    // eliminar bajo mi autorización»). Los nombres de las etapas, las
    // descripciones y las leyendas son el sitio público, igual que
    // sitio/publicado. Siempre con copia previa en _historial.
    // Crear y borrar los permite la regla, pero la herramienta del agente
    // (firestore.mjs, `conAutorizacion`) no lo hace sin la autorización de
    // Mauro citada, que queda escrita en la entrada del historial. Una regla
    // no puede leer un chat: esa mitad de la garantía vive en la herramienta.
    match /albums/{album} {
      allow read: if true;                 // público: alimenta el sitio
      allow create, update, delete: if con('albumes') || esAgente();

      match /fotos/{foto} {
        allow read: if true;
        allow create, update, delete: if con('albumes') || esAgente();
      }
    }

    // ---------- cálculos de taller ----------
    // Uno por cliente. NO son públicos: llevan datos de gente y medidas de
    // obra. Sólo quien tenga el permiso 'calculo' los ve y los escribe.
    match /calculos/{id} {
      allow read, create, update, delete: if con('calculo');
    }

    // ---------- consultas de clientes (27-oct-2026) ----------
    // Una por persona que pregunta por una yurta, con la conversación entera
    // (consultas.html). Llevan su contacto: privadas, con el permiso
    // 'consultas'. El agente las lee (comodín de arriba) y las crea o
    // completa para preparar las respuestas, con la copia de antes en
    // `_historial/` (datos/herramientas/firestore.mjs); borrar, no.
    match /consultas/{id} {
      allow read, create, update, delete: if con('consultas');
      allow create, update: if esAgente();

      // Las capturas y los audios de la conversación (consultas-2, 27-oct-2026,
      // Mauro: «sacar fotos de las conversaciones de WhatsApp… copiar también
      // los audios»). Como BYTES en la base y no en Cloudinary: una dirección
      // de Cloudinary es pública y acá no hay secreto para borrarla. Forma
      // fija y techo de 900 KB (un documento no pasa de 1 MiB). El agente sólo
      // los crea (lo lee el comodín de arriba); borrarlos es de una persona.
      match /adjuntos/{aid} {
        allow read, delete: if con('consultas');
        allow create: if (con('consultas') || esAgente())
          && request.resource.data.keys().hasOnly(['en', 'tipo', 'mime', 'nombre', 'datos', 'por', 'nota'])
          && request.resource.data.tipo in ['imagen', 'audio']
          && request.resource.data.mime is string
          && request.resource.data.mime.matches('(image|audio)/[a-z0-9.+-]+')
          && request.resource.data.datos is bytes
          && request.resource.data.datos.size() <= 900000;
      }
    }

    // ---------- el sitio ----------
    // sitio/publicado ES el sitio: lo que se escribe acá lo ve el público en
    // el acto. No hay paso de revisión: quien tiene 'contenido', publica.
    //
    // sitio/taller es el SITIO PILOTO. Mismo archivo index.html, otro
    // documento: con ?taller=1 la página se pinta con éste. Existe para que
    // alguien de afuera —un diseñador— pueda mover, apagar y reescribir sin
    // poder tocar lo publicado.
    //
    // OJO CON EL ORDEN: en Firestore los permisos se SUMAN y los dos bloques
    // de abajo matchean sitio/taller. Por eso quien tiene 'contenido' puede
    // escribir los dos, y quien tiene sólo 'taller' puede escribir el taller
    // y NO publicado —que sólo matchea el bloque general—. Esa asimetría es
    // toda la garantía del piloto, y vive acá, no en la interfaz.
    //
    // Lectura abierta en los dos: el taller no guarda nada sensible, son los
    // mismos textos del catálogo ordenados de otra manera, y que se lea sin
    // cuenta es justamente lo que permite pasarle el enlace a alguien.
    // `esAgente()` entra SÓLO acá. Es todo el alcance de escritura que tiene
    // el agente en esta base, y la razón está en la cabecera: un pedido mal
    // leído se arregla volviendo a guardar el taller, no despublicando el
    // sitio. Que esta línea NO esté en el bloque de abajo es el punto entero.
    match /sitio/taller {
      allow read: if true;
      allow write: if con('contenido') || con('taller') || esAgente();
    }

    // Desde el 19-oct-2026 el agente escribe también `sitio/publicado` —sólo
    // ése—, con su copia previa en `_historial/` (ver la cabecera).
    match /sitio/{doc} {
      allow read: if true;
      allow write: if con('contenido') || (esAgente() && doc == 'publicado');
    }

    // ---------- el historial del agente ----------
    // Cómo estaba un documento antes de que el agente lo cambiara. Lo crea
    // SÓLO el agente, y nadie lo edita ni lo borra —ni siquiera él—. Lo lee
    // el agente por el comodín de arriba (para `deshacer`) y el admin.
    match /_historial/{id} {
      allow create: if esAgente();
      allow read: if admin();
    }

    // ---------- reportes ----------
    // Una falla vista desde el panel, escrita desde la pantalla donde se vio.
    // Tres decisiones que no son de trámite, y son las del molde de remate:
    //
    // 1 · EL `uid` TIENE QUE SER EL DE QUIEN ESCRIBE. Un reporte que puede
    //     mentir de quién vino no sirve para volver a preguntarle qué vio, y
    //     eso es la mitad de para qué existe un reporte.
    // 2 · LEERLOS ES DE ADMIN. Un reporte se escribe sin pensar en quién lo va
    //     a leer, así que puede nombrar a un cliente o un cálculo de taller.
    //     Se lo trata como lo más sensible que podría llegar a contener. El
    //     agente los LEE por el comodín de `esAgente()` de arriba y no los
    //     escribe —su única escritura en esta base es `sitio/taller`—, que es
    //     justamente para lo que existe este circuito: lee el pedido acá y lo
    //     ejecuta allá.
    // 3 · `texto` Y `uid` NO SE PUEDEN CAMBIAR. Se cierra o se marca tomado;
    //     no se reescribe. Pisar lo que dijo una persona es perder el reporte,
    //     y es justo lo que nadie nota hasta que hace falta.
    //
    // Lo escribe cualquiera con sesión ACTIVA, no hace falta un permiso: si
    // alguien entra al panel puede ver una falla, y pedirle un permiso para
    // contarla es garantizar que no la cuente.
    match /reportes/{id} {
      allow create: if activo()
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.estado == 'nuevo';

      allow get, list: if admin();

      allow update: if admin()
                    && request.resource.data.texto == resource.data.texto
                    && request.resource.data.uid == resource.data.uid;
      allow delete: if admin();
    }

    // ---------- avisos_contacto (30-sep-2026) ----------
    // El número y la clave de CallMeBot de cada persona, y `agente`: si
    // quiere que Claude le escriba. Es de ella y de nadie más —acá, a
    // diferencia de Casa Verde, ninguna pantalla le avisa a otro desde el
    // navegador, así que el resto del equipo no lo necesita leer—. Una clave
    // de CallMeBot sólo sirve para mandarle mensajes al número que la generó.
    // El agente: `get` de a uno, para mandar un aviso. Nunca `list`, nunca
    // escribir: el consentimiento lo da la persona.
    match /avisos_contacto/{uid} {
      allow get: if (activo() && request.auth.uid == uid) || esAgente();
      allow create, update: if activo() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(
             ['telefono', 'apikey', 'nombre', 'agente', 'actualizadoEn', 'agenteEn'])
        && (!('agente' in request.resource.data) || request.resource.data.agente is bool);
      allow delete: if activo() && request.auth.uid == uid;
    }

    // ---------- avisos (9-oct-2026) ----------
    // El buzón de cada persona: lo que Claude le avisa, y que su Pizarra trae.
    match /avisos/{id} {
      allow read, delete: if activo() && resource.data.uid == request.auth.uid;
      allow update: if activo() && resource.data.uid == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['leido']);
      allow create: if esAgente()
        && request.resource.data.keys().hasOnly(['uid', 'sitio', 'tema', 'texto', 'creadoEn', 'leido'])
        && request.resource.data.texto is string
        && request.resource.data.texto.size() <= 600;
    }

    // ---------- todo lo demás, cerrado ----------
    // Ojo con leer esto como un candado: en Firestore los permisos se SUMAN y
    // un `if false` no le resta nada a lo de arriba. Lo que cierra de verdad
    // es el deny por defecto — una colección sin bloque propio es inaccesible
    // aunque este bloque no existiera. Queda escrito como recordatorio.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
